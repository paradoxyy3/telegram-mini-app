    <script>
        const tg = window.Telegram.WebApp;
        tg.expand(); // Разворачиваем окно на весь экран

        // ⚠️ Укажи IP/домен своего сервера где запущен Python бот
        const API_URL = "http://ТВОЙ_ИП_СЕРВЕРА:8080/api/active_admins";

        async function loadAdmins() {
            try {
                const res = await fetch(API_URL);
                const admins = await res.json();
                const container = document.getElementById('admins-list');
                container.innerHTML = '';

                if (admins.length === 0) {
                    container.innerHTML = '<div style="text-align:center; padding:20px;">😴 Сейчас нет активных админов</div>';
                    return;
                }

                admins.forEach(adm => {
                    const card = document.createElement('div');
                    card.className = 'card';
                    card.onclick = () => openModal(adm);

                    const verifyHtml = adm.is_verified ? '<div class="verify-badge">✓</div>' : '';
                    const avatarSrc = adm.avatar ? adm.avatar : 'https://via.placeholder.com/150';

                    card.innerHTML = `
                        <div class="avatar-container">
                            <img class="avatar" src="${avatarSrc}">
                            <div class="online-badge"></div>
                        </div>
                        <div class="card-info">
                            <div class="admin-title">${adm.prefix}</div>
                            <div class="admin-desc">${adm.info}</div>
                            <div class="last-active">Последняя активность ${adm.mins_ago} мин. назад</div>
                        </div>
                        ${verifyHtml}
                    `;
                    container.appendChild(card);
                });
            } catch (err) {
                console.error("Ошибка загрузки:", err);
            }
        }

        function openModal(adm) {
            document.getElementById('modal-img').src = adm.avatar || 'https://via.placeholder.com/150';
            document.getElementById('modal-name').innerText = adm.prefix;
            document.getElementById('modal-desc').innerText = adm.info;
            document.getElementById('modal').style.display = 'flex';
        }

        function closeModal() {
            document.getElementById('modal').style.display = 'none';
        }

        loadAdmins();
    </script>
